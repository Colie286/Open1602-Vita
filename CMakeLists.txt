cmake_minimum_required(VERSION 3.25)
project(Open1602 VERSION 0.3.1 LANGUAGES C CXX)
if(PROJECT_SOURCE_DIR STREQUAL PROJECT_BINARY_DIR)
    message(FATAL_ERROR "Use a separate build directory")
endif()
if(NOT (CMAKE_CROSSCOMPILING AND CMAKE_SYSTEM_PROCESSOR STREQUAL "armv7-a" AND VITASDK))
    message(FATAL_ERROR "This source export requires the VitaSDK ARM toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(O1602_BUILD_VITA ON)
set(O1602_BUILD_GAME ON)
set(O1602_VITA_DIAGNOSTIC OFF)
add_compile_options(-fstack-usage)
add_compile_definitions(O1602_EXPERIMENTAL_VITA=1 _POSIX_C_SOURCE=200809L)
add_compile_definitions($<$<CONFIG:Debug>:_GLIBCXX_ASSERTIONS>)
list(APPEND CMAKE_MODULE_PATH "${PROJECT_SOURCE_DIR}/cmake")
include(CompilerWarnings)
include(EmbedFiles)
include(WindowsUtf8)
option(O1602_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
set(O1602_OUTPUT_DIR "${PROJECT_BINARY_DIR}/bin" CACHE PATH "Executable output directory")
function(o1602_set_output_dir target)
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${O1602_OUTPUT_DIR}$<0:>")
endfunction()
find_package(ZLIB REQUIRED)
add_subdirectory(third_party/enet)
add_subdirectory(third_party/stb_vorbis)
add_subdirectory(third_party/dr_flac)
add_subdirectory(third_party/lua)
set_target_properties(o1602_enet o1602_lua o1602_stb_vorbis o1602_dr_flac
    PROPERTIES POSITION_INDEPENDENT_CODE OFF)
add_subdirectory(engine)
add_subdirectory(platform/vita)
